Gauss-Seidel relaxation（高斯-赛德尔松弛）
求出上一步中的x数组值
```cpp
void diffuse( int N, int b, float* x, float* x0, float diff, float dt)
{ 
    int i, j, k; 
    float a = dt*diff*N*N;
    for( k = 0; k < 20; k++) 
    { 
        //对grid进行20次循环 求逼近值
        for( i = 1; i <= N; i++) 
        { 
            for( j=1; j<=N; j++) 
            {
                x[IX(i,j)] = (
                    x0[IX(i,j)] + 
                    a * (
                        x[IX(i-1,j)] +
                        x[IX(i+1,j)] + 
                        x[IX(i,j-1)] + 
                        x[IX(i,j+1)])
                        ) / (1 + 4*a);
            }
        }
        set_bnd( N, b, x);
    } 
}
```

```cpp
//平流(对流)层计算
//u0,v0=力场
void advect( int N, int b, float* d, float* d0, float* u, float* v, float dt) {
    int i, j, i0, j0, i1, j1; 
    float x, y, s0, t0, s1, t1, dt0; 
    dt0 = dt * N;
    for( i = 1; i <= N; i++) { 
        for( j = 1; j<= N; j++) {

            x= i - dt0 * u[IX(i,j)]; 
            y= j - dt0 * v[IX(i,j)]; 
            if(x < 0.5) x = 0.5; 
            if(x > N + 0.5) x = N + 0.5; 
            i0 = (int)x; 
            i1 = i0+1; 
            if(y < 0.5) y = 0.5; 
            if(y > N + 0.5) y = N + 0.5; 
            j0 = (int)y; 
            j1 = j0 + 1; 
            s1 = x - i0; 
            s0 = 1 - s1; 
            t1 = y - j0; 
            t0 = 1 - t1; 
            d[IX(i,j)]= s0 * (t0 * d0[IX(i0,j0)] + t1 * d0[IX(i0,j1)]) + 
            s1 * (t0 * d0[IX(i1,j0)] + t1 * d0[IX(i1,j1)]);

        }
    }
    set_bnd( N, b, d);
}



//x0 源密度 source densities
void dens_step( int N, float* x, float* x0, float* u, float* v, float diff, float dt) {
    //往x数组中 添加力 x0是额外输入的力
    add_source( N, x, x0, dt); 
	//交换 x0变成了 添加力之后的结果
    SWAP( x0, x); 
    //计算每个格子和周围格子的密度差 净差
	//通过计算x0得出散射后的x
    diffuse( N, 0, x, x0, diff, dt); 
	//再次交换 x0是上一次散射后的结果
	//x成为这次输出结果
    SWAP( x0, x); 
    advect( N, 0, x, x0, u, v, dt);
}

//速度更新
void vel_step ( int N, float * u, float * v, float * u0, float * v0, float visc, float dt )
{
    add_source ( N, u, u0, dt ); 
	add_source ( N, v, v0, dt );
    
	SWAP ( u0, u ); 
	diffuse ( N, 1, u, u0, visc, dt );
    
	SWAP ( v0, v ); 
	diffuse ( N, 2, v, v0, visc, dt );
    
	project ( N, u, v, u0, v0 );

    SWAP ( u0, u ); 
	SWAP ( v0, v );

    advect ( N, 1, u, u0, u0, v0, dt ); 	
	advect ( N, 2, v, v0, u0, v0, dt );

    project ( N, u, v, u0, v0 );
}

//速度场=质量守恒场+一个梯度场
//质量守恒场 漂亮的旋涡状涡旋
void project ( int N, float * u, float * v, float * p, float * div )
{
    int i, j, k;
    float h;
    h = 1.0/N;
    for ( i=1 ; i<=N ; i++ ) 
	{
        for ( j=1 ; j<=N ; j++ ) 
		{
            div[IX(i,j)] = -0.5 * h *(
							u[IX(i+1,j)]-u[IX(i-1,j)]+
                            v[IX(i,j+1)]-v[IX(i,j-1)]);
            p[IX(i,j)] = 0;
        }
    }

    set_bnd ( N, 0, div ); 
	set_bnd ( N, 0, p );

    for ( k=0 ; k<20 ; k++ ) 
	{
        for ( i=1 ; i<=N ; i++ ) 
		{
            for ( j=1 ; j<=N ; j++ ) 
			{
                p[IX(i,j)] = (div[IX(i,j)]+
							p[IX(i-1,j)]+p[IX(i+1,j)]+
                            p[IX(i,j-1)]+p[IX(i,j+1)])/4;
            }
        }
        set_bnd ( N, 0, p );
    }

    for ( i=1 ; i<=N ; i++ ) 
	{
        for ( j=1 ; j<=N ; j++ ) 
		{
            u[IX(i,j)] -= 0.5*(p[IX(i+1,j)]-p[IX(i-1,j)])/h;
            v[IX(i,j)] -= 0.5*(p[IX(i,j+1)]-p[IX(i,j-1)])/h;
        }
    }

    set_bnd ( N, 1, u ); 
	set_bnd ( N, 2, v );
}
```

## 基礎知識
### 梯度
梯度是一个向量,表示变化率 梯度: 密度的梯度变化: 梯度点乘梯度 二阶导之和
### 散度(Divergence)
梯度x速度: 向外涌出: 向内吸入:

### 旋度(Curl)
梯度和速度u进行“叉乘”运算时结果是旋度 流体里的每一个点绕着自身打转的趋势有多强。就像你把小纸船放在水里，船转圈的速度就取决于该点的旋度

### 拉普拉斯算子(Laplacian)
函数 一阶导 是斜率 二阶导 是斜率的变化率 二阶导 > 0 凹点,周围值比当前大 二阶导 < 0 凸点,周围值比当前小
得出这个点的值比周围邻居高还是低 代码实现
```cpp
void set_bnd ( int N, int b, float * x )
{
    int i;
    for ( i=1 ; i<=N ; i++ ) 
    {//更新周围墙壁
        x[IX(0 , i)] = b == 1 ? -x[IX(1,i)] : x[IX(1,i)];//上
        x[IX(N+1, i)] = b == 1 ? -x[IX(N,i)] : x[IX(N,i)];//下
        x[IX(i , 0)] = b==2 ? -x[IX(i,1)] : x[IX(i,1)];//左
        x[IX(i , N+1)] = b==2 ? -x[IX(i,N)] : x[IX(i,N)];//右
    }
    //四个角
    x[IX(0 , 0 )] = 0.5 * (x[IX(1,0 )] + x[IX(0 ,1)]);
    x[IX(0 , N+1)] = 0.5 * (x[IX(1,N+1)] + x[IX(0 ,N)]);
    x[IX(N+1, 0 )] = 0.5 * (x[IX(N,0 )] + x[IX(N+1,1)]);
    x[IX(N+1, N+1)] = 0.5 * (x[IX(N,N+1)] + x[IX(N+1,N)]);
}
```
