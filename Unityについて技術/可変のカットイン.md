float Parallelogram(float2 uv,float upSlope,float downSlope,float upOffset,float downOffset)
{
	float2 pos = uv - float2(0.5,0.5);	
	float upLine = tan(upSlope) * pos.x + upOffset;
	
	float downLine = tan(downSlope) * pos.x + downOffset;
	float sdfx = 0.5f - abs(pos.x);
	
	return min(min(pos.y - downLine, upLine - pos.y),sdfx);
}

void Parallelogram_float(float2 uv,float upSlope,float downSlope,float upOffset,float downOffset,out float sdf)
{
	sdf = Parallelogram(uv,upSlope,downSlope,upOffset,downOffset);
}

  

void Parallelogram_half(float2 uv,float upSlope,float downSlope,float upOffset,float downOffset,out half sdf)
{
	sdf = Parallelogram(uv,upSlope,downSlope,upOffset,downOffset);
}
