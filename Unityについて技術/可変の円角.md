float TriangleWithRoundTip(float2 uv, float input_radians, float input_radius)
{

float2 p = uv - float2(0.5, 0.5);

float r = input_radius;
float s = max(sin(input_radians), 1e-5);
float c = cos(input_radians);

float lenP = max(length(p), 1e-5);
float2 dir = p / lenP;

float cosValue = dot(dir, float2(1.0, 0.0));

float triangleMask = step(cos(input_radians), cosValue);

float circleX = r / s;
float2 circleCenter = float2(circleX, 0.0);

  
float distToCircle = length(p - circleCenter);
float circleMask = 1.0 - step(r, distToCircle);

float tangentX = circleX * c * c;
float cutMask = step(tangentX, p.x);
float triangleBodyMask = triangleMask * cutMask;

float mask = max(triangleBodyMask, circleMask);
return mask;
}
